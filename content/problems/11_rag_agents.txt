import torch


def top_k_retrieve(query: torch.Tensor, docs: torch.Tensor, k: int):
    q = query / query.norm()
    d = docs / docs.norm(dim=1, keepdim=True)
    sims = d @ q
    k = min(k, docs.shape[0])
    scores, indices = sims.topk(k)
    return indices, scores

=== bm25 | BM25 Scoring | Medium
--- description
Implement `bm25_scores(query, docs, k1=1.5, b=0.75)`.

`query` is a list of tokens and `docs` a list of tokenized documents. For each document `d`, sum over the distinct query terms `t`:

```
idf(t)   = ln((N - df(t) + 0.5) / (df(t) + 0.5) + 1)
score(d) = sum_t idf(t) * tf(t, d) * (k1 + 1) / (tf(t, d) + k1 * (1 - b + b * len(d) / avgdl))
```

`N` is the number of documents, `df(t)` the number of documents containing `t`, `tf(t, d)` the count of `t` in `d`, and `avgdl` the average document length. Return a float tensor of shape `(N,)`.

import math
import torch


def bm25_scores(query: list, docs: list, k1: float = 1.5, b: float = 0.75) -> torch.Tensor:
    n = len(docs)
    avgdl = sum(len(d) for d in docs) / n
    scores = torch.zeros(n)
    for term in set(query):
        df = sum(1 for d in docs if term in d)
        idf = math.log((n - df + 0.5) / (df + 0.5) + 1)
        for i, d in enumerate(docs):
            tf = d.count(term)
            if tf == 0:
                continue
            scores[i] += idf * tf * (k1 + 1) / (tf + k1 * (1 - b + b * len(d) / avgdl))
    return scores

=== mmr | Maximal Marginal Relevance | Medium
--- description
Implement `mmr(query, docs, k, lam)` to select documents that are relevant but not redundant.

- `query`: `(D,)`, `docs`: `(N, D)`; use cosine similarity throughout
- select greedily. At each step choose the unselected document maximizing

```
lam * sim(query, d) - (1 - lam) * max over selected s of sim(d, s)
```

(the second term is 0 while nothing is selected). Break ties by the lowest index.

Return the selected indices as a list of ints, in selection order, of length `min(k, N)`.

import torch


def mmr(query: torch.Tensor, docs: torch.Tensor, k: int, lam: float = 0.5) -> list:
    d = docs / docs.norm(dim=1, keepdim=True)
    q = query / query.norm()
    rel = d @ q
    sim = d @ d.T
    selected = []
    remaining = list(range(docs.shape[0]))
    while remaining and len(selected) < k:
        best, best_score = None, None
        for i in remaining:
            redundancy = max(sim[i, j].item() for j in selected) if selected else 0.0
            score = lam * rel[i].item() - (1 - lam) * redundancy
            if best_score is None or score > best_score + 1e-9:
                best, best_score = i, score
        selected.append(best)
        remaining.remove(best)
    return selected

def chunk_tokens(tokens: list, chunk_size: int, overlap: int = 0) -> list:
    chunks = []
    step = chunk_size - overlap
    start = 0
    while start < len(tokens):
        chunks.append(tokens[start:start + chunk_size])
        if start + chunk_size >= len(tokens):
            break
        start += step
    return chunks

def reciprocal_rank_fusion(rankings: list, k: int = 60) -> list:
    scores = {}
    for ranking in rankings:
        for rank, doc in enumerate(ranking, start=1):
            scores[doc] = scores.get(doc, 0.0) + 1.0 / (k + rank)
    return sorted(scores, key=lambda d: (-scores[d], d))

def recall_at_k(retrieved: list, relevant, k: int) -> float:
    relevant = set(relevant)
    if not relevant:
        return 0.0
    return len(relevant & set(retrieved[:k])) / len(relevant)


def mean_reciprocal_rank(results: list) -> float:
    if not results:
        return 0.0
    total = 0.0
    for retrieved, relevant in results:
        relevant = set(relevant)
        for rank, doc in enumerate(retrieved, start=1):
            if doc in relevant:
                total += 1.0 / rank
                break
    return total / len(results)

import json
import re


def parse_tool_call(text: str):
    match = re.search(r"<tool_call>(.*?)</tool_call>", text, flags=re.S)
    if not match:
        return None
    try:
        data = json.loads(match.group(1).strip())
    except (ValueError, TypeError):
        return None
    if not isinstance(data, dict):
        return None
    name = data.get("name")
    if not isinstance(name, str):
        return None
    arguments = data.get("arguments", {})
    if not isinstance(arguments, dict):
        return None
    return {"name": name, "arguments": arguments}

=== agent-loop | The Agent Loop | Hard
--- description
Implement the core loop of a tool-using agent.

`run_agent(llm, tools, user_message, max_steps)`:

- `llm(messages)` returns either `{"type": "final", "content": str}` or `{"type": "tool_call", "name": str, "arguments": dict}`
- `tools` maps tool names to Python callables invoked as `tool(**arguments)`

Start with `messages = [{"role": "user", "content": user_message}]` and repeat up to `max_steps` times:

1. call `llm(messages)`
2. if it is final, append `{"role": "assistant", "content": content}` and return `(content, messages)`
3. otherwise append `{"role": "assistant", "tool_call": {"name": name, "arguments": arguments}}`, run the tool, and append `{"role": "tool", "name": name, "content": result}` where `result` is `str(return value)`

Tool failures must not crash the loop; report them to the model instead:

- unknown tool name: `result = "error: unknown tool <name>"`
- the tool raises an exception `e`: `result = "error: <str(e)>"`

If the model has not produced a final answer after `max_steps` LLM calls, return `(None, messages)`.

def run_agent(llm, tools: dict, user_message: str, max_steps: int = 5):
    messages = [{"role": "user", "content": user_message}]
    for _ in range(max_steps):
        reply = llm(messages)
        if reply["type"] == "final":
            messages.append({"role": "assistant", "content": reply["content"]})
            return reply["content"], messages
        name, arguments = reply["name"], reply.get("arguments", {})
        messages.append({"role": "assistant", "tool_call": {"name": name, "arguments": arguments}})
        if name not in tools:
            result = "error: unknown tool %s" % name
        else:
            try:
                result = str(tools[name](**arguments))
            except Exception as exc:
                result = "error: %s" % exc
        messages.append({"role": "tool", "name": name, "content": result})
    return None, messages
--- tests
def _scripted(replies):
    state = {"i": 0, "seen": []}
    def llm(messages):
        state["seen"].append([dict(m) for m in messages])
        reply = replies[min(state["i"], len(replies) - 1)]
        state["i"] += 1
        return reply
    return llm, state

def test_direct_answer():
    """No tool needed"""
    llm, state = _scripted([{"type": "final", "content": "hi"}])
    answer, messages = run_agent(llm, {}, "hello")

    check_equal(messages, [{"role": "user", "content": "hello"}, {"role": "assistant", "content": "hi"}], name="messages")

=== trim-history | Trimming Conversation History | Medium
--- description
Long conversations must be cut to fit a context budget. Implement `trim_history(messages, max_tokens, count_tokens)`.

- never split a message, and keep the original order

--- solution
def trim_history(messages: list, max_tokens: int, count_tokens) -> list:
    system = []
    rest = messages
    if messages and messages[0].get("role") == "system":
        system, rest = [messages[0]], messages[1:]
    budget = max_tokens - sum(count_tokens(m) for m in system)
    kept = []
    for m in reversed(rest):
        cost = count_tokens(m)
        if cost > budget:
            break
        kept.append(m)
        budget -= cost
    return system + kept[::-1]

=== pass-hat-k | Agent Reliability: pass@k and pass^k | Medium
--- description
Agent benchmarks run every task several times. From `n` trials of a task with `c` successes, two metrics are estimated for a smaller number of attempts `k <= n`:

- **pass@k**: the probability that at least one of `k` attempts succeeds: `1 - C(n - c, k) / C(n, k)`
- **pass^k**: the probability that all `k` attempts succeed: `C(c, k) / C(n, k)`

where `C(a, b)` is the binomial coefficient (0 when `b > a`).

Implement `pass_metrics(results, k)` where `results` is a list of lists of booleans (one inner list of trial outcomes per task; all tasks have the same number of trials `n >= k`). Return `(pass_at_k, pass_hat_k)`, each averaged over tasks, as floats.

import math


def pass_metrics(results: list, k: int):
    at_k, hat_k = 0.0, 0.0
    for trials in results:
        n, c = len(trials), sum(1 for t in trials if t)
        total = math.comb(n, k)
        at_k += 1.0 - math.comb(n - c, k) / total
        hat_k += math.comb(c, k) / total
    return at_k / len(results), hat_k / len(results)

def test_k_one_is_success_rate():
    """For k = 1 both metrics equal the average success rate"""
    results = [[True, False, True, True], [False, False, True, False]]
    at1, hat1 = pass_metrics(results, 1)
    check_close(at1, (0.75 + 0.25) / 2, name="pass@1")
    check_close(hat1, (0.75 + 0.25) / 2, name="pass^1")

def test_hand_example():
    """n = 4 trials, c = 2 successes, k = 2"""
    at2, hat2 = pass_metrics([[True, True, False, False]], 2)
    check_close(at2, 1 - 1 / 6, name="pass@2")
    check_close(hat2, 1 / 6, name="pass^2")

def test_extremes():
    """Always-passing and always-failing tasks"""
    at3, hat3 = pass_metrics([[True] * 5, [False] * 5], 3)
    check_close(at3, 0.5, name="pass@3")
    check_close(hat3, 0.5, name="pass^3")

def test_reliability_gap():
    """pass^k falls and pass@k rises as k grows"""
    results = [[True, True, True, False, True, False, True, True]] * 3
    values = [pass_metrics(results, k) for k in (1, 2, 4, 8)]
    check_true(all(values[i][0] <= values[i + 1][0] + 1e-12 for i in range(3)), "pass@k should not decrease with k")
    check_true(all(values[i][1] >= values[i + 1][1] - 1e-12 for i in range(3)), "pass^k should not increase with k")

    check_close(values[3][1], 0.0, name="pass^8")

def test_k_equals_n():
    """With k = n, pass^k is 1 only for a perfect task"""
    at, hat = pass_metrics([[True, True, True], [True, True, False]], 3)
    check_close(hat, 0.5, name="pass^3")
