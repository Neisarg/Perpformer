import torch


def greedy_decode(model, ids: torch.Tensor, max_new_tokens: int, eos_id=None) -> torch.Tensor:
    with torch.no_grad():
        for _ in range(max_new_tokens):
            logits = model(ids)
            next_id = logits[:, -1].argmax(dim=-1, keepdim=True)
            ids = torch.cat([ids, next_id], dim=1)
            if eos_id is not None and next_id.item() == eos_id:
                break
    return ids

import torch


def top_k_filter(logits: torch.Tensor, k: int) -> torch.Tensor:
    if k >= logits.shape[-1]:
        return logits.clone()
    kth = logits.topk(k, dim=-1).values[..., -1:]
    return logits.masked_fill(logits < kth, float("-inf"))

=== top-p-filter | Top-p (Nucleus) Filtering | Medium
--- description
Implement `top_p_filter(logits, p)` for logits of shape `(V,)` or `(B, V)`.

Sort tokens by probability (descending). Keep the smallest prefix whose cumulative probability reaches `p`: a token is removed when the cumulative probability of the tokens before it is already `>= p`. The most likely token is always kept. Removed tokens get a logit of `-inf`; kept tokens keep their original logits, in their original positions.

import torch


def top_p_filter(logits: torch.Tensor, p: float) -> torch.Tensor:
    sorted_logits, sorted_idx = torch.sort(logits, dim=-1, descending=True)
    probs = torch.softmax(sorted_logits, dim=-1)
    cum_before = probs.cumsum(dim=-1) - probs
    remove_sorted = cum_before >= p
    remove_sorted[..., 0] = False
    remove = torch.zeros_like(remove_sorted).scatter(-1, sorted_idx, remove_sorted)
    return logits.masked_fill(remove, float("-inf"))

import torch


def sample_next_token(logits: torch.Tensor, temperature: float = 1.0, top_k=None, generator=None) -> torch.Tensor:
    if temperature == 0:
        return logits.argmax(dim=-1)
    logits = logits / temperature
    if top_k is not None and top_k < logits.shape[-1]:
        kth = logits.topk(top_k, dim=-1).values[..., -1:]
        logits = logits.masked_fill(logits < kth, float("-inf"))
    probs = torch.softmax(logits, dim=-1)
    return torch.multinomial(probs, num_samples=1, generator=generator).squeeze(-1)

=== beam-search | Beam Search | Hard
--- description
Implement `beam_search(log_prob_fn, start_id, beam_width, max_len, eos_id)`.

- `log_prob_fn(seq)` takes a list of token ids and returns a 1-D tensor of next-token log-probabilities of size `V`
- start with one hypothesis `[start_id]` with score 0
- at each step, extend every unfinished hypothesis with every token, adding the token's log-probability to its score; hypotheses that already end in `eos_id` are carried over unchanged
- keep the `beam_width` highest-scoring hypotheses
- stop when every kept hypothesis ends in `eos_id`, or after `max_len` steps

Return `(best_sequence, best_score)`: the highest-scoring hypothesis as a list of ints (including `start_id`) and its total log-probability as a float. No length normalization.

import torch


def beam_search(log_prob_fn, start_id: int, beam_width: int, max_len: int, eos_id: int):
    beams = [([start_id], 0.0)]
    for _ in range(max_len):
        if all(seq[-1] == eos_id and len(seq) > 1 for seq, _ in beams):
            break
        candidates = []
        for seq, score in beams:
            if len(seq) > 1 and seq[-1] == eos_id:
                candidates.append((seq, score))
                continue
            log_probs = log_prob_fn(seq)
            for tok in range(log_probs.shape[0]):
                candidates.append((seq + [tok], score + log_probs[tok].item()))
        candidates.sort(key=lambda c: c[1], reverse=True)
        beams = candidates[:beam_width]
    best_seq, best_score = max(beams, key=lambda c: c[1])
    return best_seq, best_score

=== bpe | Byte-Pair Encoding | Hard
--- description
Implement the core of BPE tokenizer training and encoding.

`bpe_train(word_freqs, num_merges)`:

def _merge(symbols, pair):
    out, i = [], 0
    while i < len(symbols):
        if i + 1 < len(symbols) and symbols[i] == pair[0] and symbols[i + 1] == pair[1]:
            out.append(symbols[i] + symbols[i + 1])
            i += 2
        else:
            out.append(symbols[i])
            i += 1
    return out


def bpe_train(word_freqs: dict, num_merges: int) -> list:
    words = {w: list(w) for w in word_freqs}
    merges = []
    for _ in range(num_merges):
        counts = {}
        for w, symbols in words.items():
            for a, b in zip(symbols[:-1], symbols[1:]):
                counts[(a, b)] = counts.get((a, b), 0) + word_freqs[w]
        if not counts:
            break
        best = min(counts, key=lambda p: (-counts[p], p))
        merges.append(best)
        words = {w: _merge(s, best) for w, s in words.items()}
    return merges


def bpe_encode(word: str, merges: list) -> list:
    symbols = list(word)
    for pair in merges:
        symbols = _merge(symbols, pair)
    return symbols

import torch
import torch.nn.functional as F


def perplexity(logits: torch.Tensor, targets: torch.Tensor, pad_id=None) -> float:
    nll = F.cross_entropy(logits, targets, reduction="none")
    if pad_id is not None:
        nll = nll[targets != pad_id]
    return torch.exp(nll.mean()).item()

import math
import torch


def tfidf(docs: list):
    vocab = sorted({t for d in docs for t in d})
    index = {t: i for i, t in enumerate(vocab)}
    n = len(docs)
    tf = torch.zeros(n, len(vocab))
    for i, d in enumerate(docs):
        for t in d:
            tf[i, index[t]] += 1.0
        if len(d) > 0:
            tf[i] /= len(d)
    df = (tf > 0).sum(dim=0).float()
    idf = torch.log(n / df)
    return tf * idf, vocab

=== edit-distance | Edit Distance and Word Error Rate | Medium
--- description

--- solution
def edit_distance(a, b) -> int:
    prev = list(range(len(b) + 1))
    for i in range(1, len(a) + 1):
        cur = [i] + [0] * len(b)
        for j in range(1, len(b) + 1):
            cost = 0 if a[i - 1] == b[j - 1] else 1
            cur[j] = min(prev[j] + 1, cur[j - 1] + 1, prev[j - 1] + cost)
        prev = cur
    return prev[len(b)]


def wer(reference: str, hypothesis: str) -> float:
    ref, hyp = reference.split(), hypothesis.split()
    return edit_distance(ref, hyp) / len(ref)
--- tests
def test_classic():
    """kitten -> sitting needs 3 edits"""
    check_equal(edit_distance("kitten", "sitting"), 3)
    check_equal(edit_distance("flaw", "lawn"), 2)

def test_empty_and_identical():
    """Empty strings and identical strings"""
    check_equal(edit_distance("", ""), 0)
    check_equal(edit_distance("abc", ""), 3)
    check_equal(edit_distance("", "abcd"), 4)
    check_equal(edit_distance("same", "same"), 0)

def test_lists():
    """Works on lists of tokens"""
    check_equal(edit_distance(["a", "b", "c"], ["a", "x", "c", "d"]), 2)

def test_wer():
    """One substitution and one deletion over six reference words"""
    check_close(wer("the cat sat on the mat", "the cat sit on mat"), 2 / 6)
    check_close(wer("hello world", "hello world"), 0.0)

def test_wer_above_one():
    """Many insertions can push WER above 1"""
    check_close(wer("hi", "well hi there my friend"), 4.0)
