import torch


def ridge_regression(X: torch.Tensor, y: torch.Tensor, lam: float = 0.0) -> torch.Tensor:
    d = X.shape[1]
    A = X.T @ X + lam * torch.eye(d, dtype=X.dtype)
    return torch.linalg.solve(A, X.T @ y)

import torch


def linear_regression_gd(X: torch.Tensor, y: torch.Tensor, lr: float = 0.1, steps: int = 100):
    n, d = X.shape
    w = torch.zeros(d, dtype=X.dtype)
    b = torch.zeros((), dtype=X.dtype)
    for _ in range(steps):
        residual = X @ w + b - y
        grad_w = 2.0 / n * (X.T @ residual)
        grad_b = 2.0 * residual.mean()
        w = w - lr * grad_w
        b = b - lr * grad_b
    return w, b

def predict_proba(X: torch.Tensor, w: torch.Tensor, b) -> torch.Tensor:

def logistic_regression_fit(X: torch.Tensor, y: torch.Tensor, lr: float = 0.1, steps: int = 100):
    n, d = X.shape
    w = torch.zeros(d, dtype=X.dtype)
    b = torch.zeros((), dtype=X.dtype)
    for _ in range(steps):
        p = predict_proba(X, w, b)
        err = p - y
        w = w - lr * (X.T @ err) / n
        b = b - lr * err.mean()
    return w, b

=== kmeans | k-Means (Lloyd's Algorithm) | Medium
--- description
Implement `kmeans(X, init_centroids, iters)`.

Repeat `iters` times:

1. assign every point to its nearest centroid (Euclidean distance; ties go to the lowest index)
2. move each centroid to the mean of its assigned points; a centroid with no points keeps its position

Finally compute the assignments for the final centroids.

- `X`: `(N, D)`, `init_centroids`: `(K, D)`
- returns `(centroids, assignments)` with shapes `(K, D)` and `(N,)` (integer tensor)

Do not modify `init_centroids` in place.

import torch


def kmeans(X: torch.Tensor, init_centroids: torch.Tensor, iters: int = 10):
    centroids = init_centroids.clone()
    k = centroids.shape[0]

    def assign(c):
        d = (X[:, None, :] - c[None, :, :]).pow(2).sum(dim=-1)
        return d.argmin(dim=1)

    for _ in range(iters):
        a = assign(centroids)
        for j in range(k):
            members = X[a == j]
            if members.shape[0] > 0:
                centroids[j] = members.mean(dim=0)
    return centroids, assign(centroids)

import torch


def knn_predict(X_train: torch.Tensor, y_train: torch.Tensor, X_test: torch.Tensor, k: int = 3) -> torch.Tensor:
    d = (X_test[:, None, :] - X_train[None, :, :]).pow(2).sum(dim=-1)
    idx = d.topk(k, dim=1, largest=False).indices
    votes = y_train[idx]
    num_classes = int(y_train.max().item()) + 1
    preds = []
    for row in votes:
        counts = torch.bincount(row, minlength=num_classes)
        best = counts.max()
        preds.append(int((counts == best).nonzero()[0].item()))
    return torch.tensor(preds, dtype=torch.long)

import torch


def pca(X: torch.Tensor, k: int):
    Xc = X - X.mean(dim=0, keepdim=True)
    U, S, Vh = torch.linalg.svd(Xc, full_matrices=False)
    components = Vh[:k]
    projected = Xc @ components.T
    var = S ** 2
    explained_ratio = var[:k] / var.sum()
    return components, projected, explained_ratio

import math
import torch


def gaussian_nb_predict(X_train, y_train, X_test, num_classes: int, eps: float = 1e-6) -> torch.Tensor:
    scores = []
    for c in range(num_classes):
        Xc = X_train[y_train == c]
        prior = Xc.shape[0] / X_train.shape[0]
        mean = Xc.mean(dim=0)
        var = ((Xc - mean) ** 2).mean(dim=0) + eps
        log_lik = -0.5 * (torch.log(2 * math.pi * var) + (X_test - mean) ** 2 / var).sum(dim=1)
        scores.append(math.log(prior) + log_lik)
    return torch.stack(scores, dim=1).argmax(dim=1)

=== best-split | Best Split by Gini Impurity | Medium
--- description

import torch


def best_split(x: torch.Tensor, y: torch.Tensor):
    def gini(labels):
        if labels.numel() == 0:
            return 0.0
        p1 = labels.double().mean().item()
        return 1.0 - p1 ** 2 - (1.0 - p1) ** 2

    values = torch.unique(x).tolist()
    n = x.numel()
    best_t, best_g = None, None
    for lo, hi in zip(values[:-1], values[1:]):
        t = (lo + hi) / 2.0
        left, right = y[x <= t], y[x > t]
        g = (left.numel() * gini(left) + right.numel() * gini(right)) / n
        if best_g is None or g < best_g - 1e-12:
            best_t, best_g = t, g
    return best_t, best_g

import torch


def kfold_indices(n: int, k: int):
    sizes = [n // k + (1 if i < n % k else 0) for i in range(k)]
    idx = torch.arange(n)
    folds, start = [], 0
    for size in sizes:
        val = idx[start:start + size]
        train = torch.cat([idx[:start], idx[start + size:]])
        folds.append((train, val))
        start += size
    return folds

    check_equal(folds[0][1], torch.tensor([0, 1, 2]), name="fold 0 validation")
    check_equal(folds[0][0], torch.tensor([3, 4]), name="fold 0 train")
    check_equal(folds[1][1], torch.tensor([3, 4]), name="fold 1 validation")
    check_equal(folds[1][0], torch.tensor([0, 1, 2]), name="fold 1 train")

def test_uneven_sizes():
    """10 items in 3 folds gives sizes 4, 3, 3"""
    folds = kfold_indices(10, 3)
    check_equal([f[1].numel() for f in folds], [4, 3, 3], name="validation sizes")
    check_equal(folds[1][1], torch.tensor([4, 5, 6]), name="fold 1 validation")
    check_equal(folds[1][0], torch.tensor([0, 1, 2, 3, 7, 8, 9]), name="fold 1 train")

def test_partition():
    """Every index is validated exactly once and never appears in its own training set"""
    folds = kfold_indices(23, 5)
    all_val = torch.cat([v for _, v in folds])
    check_equal(torch.sort(all_val).values, torch.arange(23), name="union of validation folds")
    for train, val in folds:
        check_equal(train.numel() + val.numel(), 23, name="train + val size")
        check_true(len(set(train.tolist()) & set(val.tolist())) == 0, "train and validation overlap")
