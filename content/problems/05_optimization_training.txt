import torch


class SGD:
    def __init__(self, params, lr: float, momentum: float = 0.0, weight_decay: float = 0.0):
        self.params = list(params)
        self.lr = lr
        self.momentum = momentum
        self.weight_decay = weight_decay
        self.buffers = [None] * len(self.params)

    def zero_grad(self):
        for p in self.params:
            p.grad = None

    def step(self):
        with torch.no_grad():
            for i, p in enumerate(self.params):
                if p.grad is None:
                    continue
                g = p.grad + self.weight_decay * p
                if self.momentum != 0.0:
                    if self.buffers[i] is None:
                        self.buffers[i] = g.clone()
                    else:
                        self.buffers[i] = self.momentum * self.buffers[i] + g
                    g = self.buffers[i]
                p.sub_(self.lr * g)

def _run(make_mine, make_ref, steps=6):

    a = torch.randn(4, 3, requires_grad=True)
    b = torch.randn(3, requires_grad=True)
    a2, b2 = a.detach().clone().requires_grad_(True), b.detach().clone().requires_grad_(True)
    mine, ref = make_mine([a, b]), make_ref([a2, b2])

            opt.zero_grad()

            loss.backward()
            opt.step()

import torch


class Adam:
    def __init__(self, params, lr: float = 1e-3, betas=(0.9, 0.999), eps: float = 1e-8):
        self.params = list(params)
        self.lr = lr
        self.beta1, self.beta2 = betas
        self.eps = eps
        self.m = [torch.zeros_like(p) for p in self.params]
        self.v = [torch.zeros_like(p) for p in self.params]
        self.t = 0

    def zero_grad(self):
        for p in self.params:
            p.grad = None

    def step(self):
        self.t += 1
        with torch.no_grad():
            for i, p in enumerate(self.params):
                if p.grad is None:
                    continue
                g = p.grad
                self.m[i] = self.beta1 * self.m[i] + (1 - self.beta1) * g
                self.v[i] = self.beta2 * self.v[i] + (1 - self.beta2) * g * g
                m_hat = self.m[i] / (1 - self.beta1 ** self.t)
                v_hat = self.v[i] / (1 - self.beta2 ** self.t)
                p.sub_(self.lr * m_hat / (v_hat.sqrt() + self.eps))

import torch


def adamw_step(param, grad, m, v, t, lr=1e-3, beta1=0.9, beta2=0.999, eps=1e-8, weight_decay=0.01):
    param = param - lr * weight_decay * param
    m = beta1 * m + (1 - beta1) * grad
    v = beta2 * v + (1 - beta2) * grad ** 2
    m_hat = m / (1 - beta1 ** t)
    v_hat = v / (1 - beta2 ** t)
    param = param - lr * m_hat / (v_hat.sqrt() + eps)
    return param, m, v

import math


def lr_at_step(step: int, warmup_steps: int, total_steps: int, base_lr: float, min_lr: float = 0.0) -> float:
    if step < warmup_steps:
        return base_lr * step / warmup_steps
    if step >= total_steps:
        return min_lr
    progress = (step - warmup_steps) / (total_steps - warmup_steps)
    return min_lr + 0.5 * (base_lr - min_lr) * (1.0 + math.cos(math.pi * progress))

import torch


def clip_grad_norm(params, max_norm: float) -> float:
    params = [p for p in params if p.grad is not None]
    total = torch.sqrt(sum((p.grad.detach() ** 2).sum() for p in params)).item()
    if total > max_norm:
        scale = max_norm / total
        for p in params:
            p.grad.detach().mul_(scale)
    return total

import torch
import torch.nn as nn


class EMA:
    def __init__(self, model: nn.Module, decay: float = 0.999):
        self.decay = decay
        self.shadow = {name: p.detach().clone() for name, p in model.named_parameters()}

    def update(self, model: nn.Module):
        with torch.no_grad():
            for name, p in model.named_parameters():
                self.shadow[name].mul_(self.decay).add_(p.detach(), alpha=1 - self.decay)

    def copy_to(self, model: nn.Module):
        with torch.no_grad():
            for name, p in model.named_parameters():
                p.copy_(self.shadow[name])

import torch
import torch.nn as nn
import torch.nn.functional as F


def train(model: nn.Module, X: torch.Tensor, y: torch.Tensor, epochs: int, lr: float, batch_size: int):
    optimizer = torch.optim.SGD(model.parameters(), lr=lr)
    history = []
    model.train()
    for _ in range(epochs):
        losses = []
        for start in range(0, X.shape[0], batch_size):
            xb, yb = X[start:start + batch_size], y[start:start + batch_size]
            optimizer.zero_grad()
            loss = F.mse_loss(model(xb), yb)
            loss.backward()
            optimizer.step()
            losses.append(loss.item())
        history.append(sum(losses) / len(losses))
    return history

import copy

def _problem():
    X = torch.randn(50, 3)
    y = X @ torch.tensor([[1.0], [-2.0], [0.5]]) + 0.3
    return torch.nn.Linear(3, 1), X, y

def _reference(model, X, y, epochs, lr, bs):
    opt = torch.optim.SGD(model.parameters(), lr=lr)
    hist = []
    for _ in range(epochs):
        losses = []
        for s in range(0, X.shape[0], bs):
            opt.zero_grad()
            loss = torch.nn.functional.mse_loss(model(X[s:s + bs]), y[s:s + bs])
            loss.backward()
            opt.step()
            losses.append(loss.item())
        hist.append(sum(losses) / len(losses))
    return hist

import torch
import torch.nn as nn


def accumulate_gradients(model: nn.Module, loss_fn, micro_batches) -> float:
    model.zero_grad()
    k = len(micro_batches)
    total = 0.0
    for x, y in micro_batches:
        loss = loss_fn(model(x), y) / k
        loss.backward()
        total += loss.item()
    return total
--- tests
def _setup():
    model = torch.nn.Sequential(torch.nn.Linear(5, 8), torch.nn.ReLU(), torch.nn.Linear(8, 2))
    X, y = torch.randn(24, 5), torch.randn(24, 2)
    return model, X, y

def test_matches_big_batch():
    """Accumulated gradients equal the full-batch gradient"""
    model, X, y = _setup()
    loss_fn = torch.nn.functional.mse_loss
    model.zero_grad()
    full = loss_fn(model(X), y)
    full.backward()
    expected = [p.grad.clone() for p in model.parameters()]
    batches = [(X[i:i + 6], y[i:i + 6]) for i in range(0, 24, 6)]
    avg = accumulate_gradients(model, loss_fn, batches)
    check_close(avg, full.item(), tol=1e-5, name="returned loss")
    for p, e in zip(model.parameters(), expected):
        check_close(p.grad, e, tol=1e-5, name="accumulated gradient")

def test_clears_stale_gradients():
    """Gradients left over from before the call are discarded"""
    model, X, y = _setup()
    loss_fn = torch.nn.functional.mse_loss
    (model(X) * 100).sum().backward()
    batches = [(X[:12], y[:12]), (X[12:], y[12:])]
    accumulate_gradients(model, loss_fn, batches)
    model2_grads = [p.grad.clone() for p in model.parameters()]
    model.zero_grad()
    loss_fn(model(X), y).backward()
    for g, p in zip(model2_grads, model.parameters()):
        check_close(g, p.grad, tol=1e-5, name="gradient")

def test_single_micro_batch():
    """One micro-batch is an ordinary backward pass"""
    model, X, y = _setup()
    loss_fn = torch.nn.functional.mse_loss
    avg = accumulate_gradients(model, loss_fn, [(X, y)])
    got = [p.grad.clone() for p in model.parameters()]
    model.zero_grad()
    full = loss_fn(model(X), y)
    full.backward()
    check_close(avg, full.item(), tol=1e-5, name="returned loss")
    for g, p in zip(got, model.parameters()):
        check_close(g, p.grad, tol=1e-5, name="gradient")
--- hints
- backward() adds to .grad. Dividing each micro-batch loss by k makes the sum equal the mean over the whole batch.

class EarlyStopping:
    def __init__(self, patience: int, min_delta: float = 0.0):
        self.patience = patience
        self.min_delta = min_delta
        self.best = float("inf")
        self.best_epoch = -1
        self.bad_epochs = 0
        self.epoch = -1

    def step(self, val_loss: float) -> bool:
        self.epoch += 1
        if val_loss < self.best - self.min_delta:
            self.best = val_loss
            self.best_epoch = self.epoch
            self.bad_epochs = 0
        else:
            self.bad_epochs += 1
        return self.bad_epochs >= self.patience
--- tests
def test_stops_after_patience():
    """Stops after `patience` epochs without improvement"""
    es = EarlyStopping(patience=2)
    results = [es.step(v) for v in [1.0, 0.8, 0.9, 0.85]]
    check_equal(results, [False, False, False, True], name="step results")
    check_close(es.best, 0.8, name="best")
    check_equal(es.best_epoch, 1, name="best_epoch")

def test_counter_resets():
    """An improvement resets the counter"""
    es = EarlyStopping(patience=2)
    results = [es.step(v) for v in [1.0, 1.1, 0.9, 1.0, 0.95, 0.5, 0.6]]
    check_equal(results, [False, False, False, False, True, False, False], name="step results")
    check_equal(es.best_epoch, 5, name="best_epoch")

def test_min_delta():
    """Improvements smaller than min_delta do not count"""
    es = EarlyStopping(patience=2, min_delta=0.1)
    results = [es.step(v) for v in [1.0, 0.95, 0.92]]
    check_equal(results, [False, False, True], name="step results")
    check_close(es.best, 1.0, name="best")

import torch
import torch.nn as nn


def param_groups(model: nn.Module, weight_decay: float):
    decay = [p for p in model.parameters() if p.requires_grad and p.dim() >= 2]
    no_decay = [p for p in model.parameters() if p.requires_grad and p.dim() < 2]
    return [
        {"params": decay, "weight_decay": weight_decay},
        {"params": no_decay, "weight_decay": 0.0},
    ]
--- tests
def _model():
    return torch.nn.Sequential(torch.nn.Embedding(10, 8), torch.nn.Linear(8, 16), torch.nn.LayerNorm(16), torch.nn.Linear(16, 4))

def test_split():
    """Matrices decay; biases and LayerNorm parameters do not"""
    groups = param_groups(_model(), 0.1)
    check_equal(len(groups), 2, name="number of groups")
    check_close(groups[0]["weight_decay"], 0.1, name="decay group weight_decay")
    check_close(groups[1]["weight_decay"], 0.0, name="no-decay group weight_decay")
    check_equal(sorted(tuple(p.shape) for p in groups[0]["params"]), [(4, 16), (10, 8), (16, 8)], name="decay shapes")
    check_equal(sorted(tuple(p.shape) for p in groups[1]["params"]), [(4,), (16,), (16,), (16,)], name="no-decay shapes")

def test_frozen_excluded():
    """Frozen parameters are left out"""
    model = _model()
    for p in model[1].parameters():
        p.requires_grad_(False)
    groups = param_groups(model, 0.1)
    total = sum(p.numel() for g in groups for p in g["params"])
    expected = sum(p.numel() for p in model.parameters() if p.requires_grad)
    check_equal(total, expected, name="number of trainable values in the groups")

def test_usable_by_optimizer():
    """The groups can be passed to AdamW"""
    opt = torch.optim.AdamW(param_groups(_model(), 0.05), lr=1e-3)
    check_equal([g["weight_decay"] for g in opt.param_groups], [0.05, 0.0], name="optimizer weight decays")
--- hints
- p.dim() >= 2 identifies weight matrices and embeddings.
- Skip parameters with requires_grad=False.
