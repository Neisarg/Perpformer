=== lora-linear | LoRA Linear Layer | Medium
--- description
Implement `LoRALinear(base, r, alpha)`, which wraps a pretrained `nn.Linear` with a low-rank adapter.

- `self.base`: the given layer, with all its parameters frozen (`requires_grad=False`)
- `self.lora_A`: `nn.Parameter` of shape `(r, in_features)`, initialized with small random values (not all zero)
- `self.lora_B`: `nn.Parameter` of shape `(out_features, r)`, initialized to zeros
- `self.scaling = alpha / r`

```
forward(x) = base(x) + scaling * (x @ lora_A.T @ lora_B.T)
```

Because `lora_B` starts at zero, the wrapped layer initially behaves exactly like the base layer.

import torch
import torch.nn as nn


class LoRALinear(nn.Module):
    def __init__(self, base: nn.Linear, r: int = 8, alpha: float = 16.0):
        super().__init__()
        self.base = base
        for p in self.base.parameters():
            p.requires_grad_(False)
        self.lora_A = nn.Parameter(torch.randn(r, base.in_features) * 0.01)
        self.lora_B = nn.Parameter(torch.zeros(base.out_features, r))
        self.scaling = alpha / r

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        return self.base(x) + self.scaling * (x @ self.lora_A.T @ self.lora_B.T)

import torch


def merge_lora(weight: torch.Tensor, lora_A: torch.Tensor, lora_B: torch.Tensor, alpha: float) -> torch.Tensor:
    r = lora_A.shape[0]
    return weight + (alpha / r) * (lora_B @ lora_A)


def unmerge_lora(merged: torch.Tensor, lora_A: torch.Tensor, lora_B: torch.Tensor, alpha: float) -> torch.Tensor:
    r = lora_A.shape[0]
    return merged - (alpha / r) * (lora_B @ lora_A)

import torch


def quantize_affine(x: torch.Tensor, num_bits: int = 8):
    qmin, qmax = 0, 2 ** num_bits - 1
    x_min = min(x.min().item(), 0.0)
    x_max = max(x.max().item(), 0.0)
    scale = (x_max - x_min) / (qmax - qmin)
    if scale == 0:
        scale = 1.0
    zero_point = int(round(qmin - x_min / scale))
    zero_point = max(qmin, min(qmax, zero_point))
    q = torch.clamp(torch.round(x / scale) + zero_point, qmin, qmax).long()
    return q, scale, zero_point


def dequantize(q: torch.Tensor, scale: float, zero_point: int) -> torch.Tensor:
    return scale * (q.float() - zero_point)

import torch


def quantize_per_channel(w: torch.Tensor, num_bits: int = 8):
    qmax = 2 ** (num_bits - 1) - 1
    max_abs = w.abs().max(dim=1).values
    scales = torch.where(max_abs > 0, max_abs / qmax, torch.ones_like(max_abs))
    q = torch.clamp(torch.round(w / scales.unsqueeze(1)), -qmax, qmax).long()
    return q, scales

    q, scales = quantize_per_channel(w, 8)

=== fake-quantize-ste | Fake Quantization with a Straight-Through Estimator | Hard
--- description
Quantization-aware training simulates quantization in the forward pass while letting gradients through.

Implement `fake_quantize(x, scale, zero_point, qmin, qmax)`:

- **forward**: `scale * (clamp(round(x / scale) + zero_point, qmin, qmax) - zero_point)`
- **backward** (straight-through estimator): the gradient with respect to `x` is 1 where the un-clamped integer `round(x / scale) + zero_point` lies in `[qmin, qmax]`, and 0 where the value was clipped

Do not use `torch.fake_quantize_per_tensor_affine`. The usual trick is `y = x + (quantized(x) - x).detach()`, combined with a mask for the clipped entries.

import torch


def fake_quantize(x: torch.Tensor, scale: float, zero_point: int, qmin: int, qmax: int) -> torch.Tensor:
    q_raw = torch.round(x / scale) + zero_point
    q = torch.clamp(q_raw, qmin, qmax)
    dq = scale * (q - zero_point)
    inside = ((q_raw >= qmin) & (q_raw <= qmax)).to(x.dtype)
    passthrough = x * inside
    return passthrough + (dq - passthrough).detach()

import torch


def magnitude_prune(weight: torch.Tensor, sparsity: float):
    k = int(sparsity * weight.numel())
    if k == 0:
        mask = torch.ones_like(weight, dtype=torch.bool)
    else:
        threshold = weight.abs().flatten().kthvalue(k).values
        mask = weight.abs() > threshold
    return weight * mask, mask

import torch
import torch.nn.functional as F


def distillation_loss(student_logits, teacher_logits, targets, T: float = 2.0, alpha: float = 0.5) -> torch.Tensor:
    log_p_student = F.log_softmax(student_logits / T, dim=-1)
    p_teacher = F.softmax(teacher_logits.detach() / T, dim=-1)
    soft = F.kl_div(log_p_student, p_teacher, reduction="batchmean")
    hard = F.cross_entropy(student_logits, targets)
    return alpha * T * T * soft + (1 - alpha) * hard
--- tests
def _ref(s, t, y, T, alpha):
    log_ps = torch.log_softmax(s / T, -1)
    pt = torch.softmax(t / T, -1)
    soft = (pt * (torch.log(pt) - log_ps)).sum(-1).mean()
    return alpha * T * T * soft + (1 - alpha) * torch.nn.functional.cross_entropy(s, y)

=== speculative-decoding | Speculative Decoding Acceptance Rule | Hard

import torch


def speculative_step(p: torch.Tensor, q: torch.Tensor, x: int, u: float):
    ratio = min(1.0, (p[x] / q[x]).item())
    if u < ratio:
        return True, None
    residual = torch.clamp(p - q, min=0)
    return False, residual / residual.sum()

import torch.nn as nn


def freeze_except(model: nn.Module, keywords) -> None:
    for name, p in model.named_parameters():
        p.requires_grad_(any(k in name for k in keywords))


def count_parameters(model: nn.Module):
    total = sum(p.numel() for p in model.parameters())
    trainable = sum(p.numel() for p in model.parameters() if p.requires_grad)
    return trainable, total
