=== cross-entropy | Cross-Entropy from Logits | Easy
--- description

- returns the mean negative log-likelihood as a scalar tensor

Compute log-probabilities stably (log-sum-exp), then pick the entry of the target class in each row.

--- solution
import torch


def cross_entropy(logits: torch.Tensor, targets: torch.Tensor) -> torch.Tensor:
    m = logits.max(dim=-1, keepdim=True).values
    log_z = m + torch.log(torch.exp(logits - m).sum(dim=-1, keepdim=True))
    log_probs = logits - log_z
    picked = log_probs[torch.arange(logits.shape[0]), targets]
    return -picked.mean()

=== bce-with-logits | Stable Binary Cross-Entropy with Logits | Medium
--- description
Implement `bce_with_logits(logits, targets)` returning the mean binary cross-entropy, where `targets` are floats in `[0, 1]` with the same shape as `logits`.

Computing `sigmoid` and then `log` overflows for large `|z|`. Use the equivalent stable form

```
loss = max(z, 0) - z * y + log(1 + exp(-|z|))
```

Do not call `F.binary_cross_entropy_with_logits` or `F.binary_cross_entropy`.

import torch


def bce_with_logits(logits: torch.Tensor, targets: torch.Tensor) -> torch.Tensor:
    loss = logits.clamp(min=0) - logits * targets + torch.log1p(torch.exp(-logits.abs()))
    return loss.mean()

    """A logit of 0 gives ln 2 for any target"""
    check_close(bce_with_logits(torch.zeros(4), torch.tensor([0.0, 1.0, 1.0, 0.0])), math.log(2.0))

def test_matches_torch():
    """Matches the built-in on random data, including soft targets"""
    z = torch.randn(6, 5) * 4
    y = torch.rand(6, 5)
    check_close(bce_with_logits(z, y), torch.nn.functional.binary_cross_entropy_with_logits(z, y))

def test_extreme_logits():
    """Finite and correct for |z| = 200"""
    z = torch.tensor([200.0, -200.0, 200.0, -200.0])
    y = torch.tensor([1.0, 0.0, 0.0, 1.0])
    out = bce_with_logits(z, y)
    check_true(torch.isfinite(out), "loss is not finite")
    check_close(out, 100.0, tol=1e-3)
--- hints
- For z >= 0 write log(1 + e^-z) + z(1 - y); for z < 0 write log(1 + e^z) - z y. The formula in the description merges both.
- torch.log1p is more accurate than log(1 + x) for small x.

import torch


def huber_loss(pred: torch.Tensor, target: torch.Tensor, delta: float = 1.0) -> torch.Tensor:
    r = (pred - target).abs()
    quadratic = 0.5 * r ** 2
    linear = delta * (r - 0.5 * delta)
    return torch.where(r <= delta, quadratic, linear).mean()

- returns a scalar tensor

--- solution
import torch
import torch.nn.functional as F


def label_smoothing_ce(logits: torch.Tensor, targets: torch.Tensor, eps: float = 0.1) -> torch.Tensor:
    log_probs = F.log_softmax(logits, dim=-1)
    nll = -log_probs.gather(1, targets.unsqueeze(1)).squeeze(1)
    uniform = -log_probs.mean(dim=-1)
    return ((1 - eps) * nll + eps * uniform).mean()
--- tests
def test_no_smoothing():
    """eps = 0 reduces to plain cross-entropy"""
    logits, targets = torch.randn(8, 5), torch.randint(0, 5, (8,))
    check_close(label_smoothing_ce(logits, targets, 0.0), torch.nn.functional.cross_entropy(logits, targets))

def test_matches_torch():
    """Matches F.cross_entropy(label_smoothing=...)"""
    logits, targets = torch.randn(12, 7) * 2, torch.randint(0, 7, (12,))
    for eps in (0.1, 0.3):
        expected = torch.nn.functional.cross_entropy(logits, targets, label_smoothing=eps)
        check_close(label_smoothing_ce(logits, targets, eps), expected, name="eps=%s" % eps)

def test_full_smoothing():
    """eps = 1 gives the cross-entropy against a uniform target"""
    logits, targets = torch.randn(4, 6), torch.randint(0, 6, (4,))
    expected = -torch.log_softmax(logits, -1).mean()
    check_close(label_smoothing_ce(logits, targets, 1.0), expected)
--- hints
- The smoothed loss is (1 - eps) * NLL(true class) + eps * mean over classes of (-log p).

=== focal-loss | Binary Focal Loss | Medium
--- description
Implement the binary focal loss from RetinaNet.

For logit `z`, label `y` in {0, 1} and `p = sigmoid(z)`:

```
p_t     = p        if y == 1 else 1 - p
alpha_t = alpha    if y == 1 else 1 - alpha
loss    = -alpha_t * (1 - p_t) ** gamma * log(p_t)
```

`focal_loss(logits, targets, gamma, alpha)` returns the mean over all elements. `targets` is a float tensor of 0s and 1s with the same shape as `logits`.

import torch
import torch.nn.functional as F


def focal_loss(logits: torch.Tensor, targets: torch.Tensor, gamma: float = 2.0, alpha: float = 0.25) -> torch.Tensor:
    ce = F.binary_cross_entropy_with_logits(logits, targets, reduction="none")
    p = torch.sigmoid(logits)
    p_t = p * targets + (1 - p) * (1 - targets)
    alpha_t = alpha * targets + (1 - alpha) * (1 - targets)
    return (alpha_t * (1 - p_t) ** gamma * ce).mean()
--- tests
def test_reduces_to_bce():
    """gamma = 0 and alpha = 0.5 gives half the BCE"""
    z, y = torch.randn(20), (torch.rand(20) > 0.5).float()
    expected = 0.5 * torch.nn.functional.binary_cross_entropy_with_logits(z, y)
    check_close(focal_loss(z, y, gamma=0.0, alpha=0.5), expected)

def test_hand_example():
    """Single positive with p = 0.5"""
    out = focal_loss(torch.tensor([0.0]), torch.tensor([1.0]), gamma=2.0, alpha=0.25)
    check_close(out, 0.25 * 0.25 * math.log(2.0))

def test_random():
    """Matches the formula on random data"""
    z, y = torch.randn(4, 9) * 2, (torch.rand(4, 9) > 0.7).float()
    p = torch.sigmoid(z)
    p_t = torch.where(y == 1, p, 1 - p)
    a_t = torch.where(y == 1, torch.tensor(0.3), torch.tensor(0.7))
    expected = (-a_t * (1 - p_t) ** 1.5 * torch.log(p_t)).mean()
    check_close(focal_loss(z, y, gamma=1.5, alpha=0.3), expected)

def test_easy_examples_downweighted():
    """A confident correct prediction contributes far less than under BCE"""
    z, y = torch.tensor([4.0]), torch.tensor([1.0])
    bce = torch.nn.functional.binary_cross_entropy_with_logits(z, y)
    check_true(focal_loss(z, y, gamma=2.0, alpha=1.0) < 0.01 * bce, "easy example was not down-weighted")
--- hints
- -log(p_t) is exactly the element-wise binary cross-entropy, so you can reuse a stable BCE with reduction="none".

import torch
import torch.nn.functional as F


def kl_divergence(p_logits: torch.Tensor, q_logits: torch.Tensor) -> torch.Tensor:
    log_p = F.log_softmax(p_logits, dim=-1)
    log_q = F.log_softmax(q_logits, dim=-1)
    return (log_p.exp() * (log_p - log_q)).sum(dim=-1).mean()

    check_close(kl_divergence(p, q), expected)

=== info-nce | Symmetric InfoNCE (CLIP) Loss | Medium
--- description
Implement the contrastive loss used by CLIP.

`a` and `b` are `(N, D)` embeddings where `a[i]` and `b[i]` are a matching pair and every other combination is a negative.

1. L2-normalize both sets of embeddings.
2. `logits = a @ b.T / temperature` (shape `(N, N)`).
3. The loss is the average of the cross-entropy over rows (a to b) and over columns (b to a), with the diagonal as the target.

Return a scalar tensor.

    # TODO
    pass
--- solution
import torch
import torch.nn.functional as F


def info_nce(a: torch.Tensor, b: torch.Tensor, temperature: float = 0.07) -> torch.Tensor:
    a = F.normalize(a, dim=-1)
    b = F.normalize(b, dim=-1)
    logits = a @ b.T / temperature
    labels = torch.arange(a.shape[0])
    return 0.5 * (F.cross_entropy(logits, labels) + F.cross_entropy(logits.T, labels))
--- tests

    """Matches a reference implementation"""

    an, bn = a / a.norm(dim=-1, keepdim=True), b / b.norm(dim=-1, keepdim=True)
    logits = an @ bn.T / 0.1
    labels = torch.arange(8)
    expected = 0.5 * (torch.nn.functional.cross_entropy(logits, labels) + torch.nn.functional.cross_entropy(logits.T, labels))
    check_close(info_nce(a, b, 0.1), expected)

def test_scale_invariant():
    """Rescaling the embeddings does not change the loss (they are normalized)"""

import torch


def triplet_loss(anchor: torch.Tensor, positive: torch.Tensor, negative: torch.Tensor, margin: float = 1.0) -> torch.Tensor:
    d_pos = (anchor - positive).pow(2).sum(dim=-1).sqrt()
    d_neg = (anchor - negative).pow(2).sum(dim=-1).sqrt()
    return (d_pos - d_neg + margin).clamp(min=0).mean()

--- solution
import torch


def precision_recall_f1(preds: torch.Tensor, targets: torch.Tensor):
    preds, targets = preds.bool(), targets.bool()
    tp = (preds & targets).sum().item()
    fp = (preds & ~targets).sum().item()
    fn = (~preds & targets).sum().item()
    precision = tp / (tp + fp) if tp + fp > 0 else 0.0
    recall = tp / (tp + fn) if tp + fn > 0 else 0.0
    f1 = 2 * precision * recall / (precision + recall) if precision + recall > 0 else 0.0
    return precision, recall, f1

=== roc-auc | ROC AUC | Medium
--- description
Implement `roc_auc(scores, labels)` returning the area under the ROC curve as a Python float.

- `scores`: `(N,)` floats, `labels`: `(N,)` of 0s and 1s
- you may assume there is at least one positive and one negative

**Example**

```
roc_auc(tensor([0.1, 0.4, 0.35, 0.8]), tensor([0, 0, 1, 1]))  ->  0.75
```

--- solution
import torch


def roc_auc(scores: torch.Tensor, labels: torch.Tensor) -> float:
    pos = scores[labels == 1]
    neg = scores[labels == 0]
    diff = pos.unsqueeze(1) - neg.unsqueeze(0)
    wins = (diff > 0).double().sum() + 0.5 * (diff == 0).double().sum()
    return (wins / (pos.numel() * neg.numel())).item()
--- tests
def test_example():
    """Example from the description"""
    check_close(roc_auc(torch.tensor([0.1, 0.4, 0.35, 0.8]), torch.tensor([0, 0, 1, 1])), 0.75)

def test_perfect_and_inverted():
    """Perfect ranking gives 1, inverted ranking gives 0"""
    labels = torch.tensor([0, 0, 1, 1, 1])
    check_close(roc_auc(torch.tensor([0.1, 0.2, 0.7, 0.8, 0.9]), labels), 1.0)
    check_close(roc_auc(torch.tensor([0.9, 0.8, 0.3, 0.2, 0.1]), labels), 0.0)

def test_ties():
    """Constant scores give 0.5"""
    check_close(roc_auc(torch.ones(6), torch.tensor([0, 1, 0, 1, 1, 0])), 0.5)

def test_random():
    """Matches a brute-force pair count"""
    scores = torch.round(torch.rand(60) * 20) / 20
    labels = (torch.rand(60) > 0.6).long()
    labels[0], labels[1] = 1, 0
    wins, total = 0.0, 0
    for i in range(60):
        for j in range(60):
            if labels[i] == 1 and labels[j] == 0:
                total += 1
                wins += 1.0 if scores[i] > scores[j] else (0.5 if scores[i] == scores[j] else 0.0)
    check_close(roc_auc(scores, labels), wins / total)
--- hints
- Compare every positive score with every negative score using broadcasting: pos[:, None] - neg[None, :].
- For large N, a rank-based formula (Mann-Whitney U) avoids the quadratic memory.

import torch


def confusion_matrix(preds: torch.Tensor, targets: torch.Tensor, num_classes: int) -> torch.Tensor:
    idx = targets.long() * num_classes + preds.long()
    return torch.bincount(idx, minlength=num_classes * num_classes).view(num_classes, num_classes)

import torch


def ndcg_at_k(relevance: torch.Tensor, k: int) -> float:
    rel = relevance.double()

    def dcg(r):
        r = r[:k]
        discounts = torch.log2(torch.arange(r.numel(), dtype=torch.float64) + 2.0)
        return ((2.0 ** r - 1.0) / discounts).sum().item()

    ideal = dcg(torch.sort(rel, descending=True).values)
    if ideal == 0:
        return 0.0
    return dcg(rel) / ideal
