=== vae-reparam-kl | VAE: Reparameterization and KL | Medium
--- description

import torch


def reparameterize(mu: torch.Tensor, logvar: torch.Tensor, eps: torch.Tensor) -> torch.Tensor:
    return mu + torch.exp(0.5 * logvar) * eps


def kl_to_standard_normal(mu: torch.Tensor, logvar: torch.Tensor) -> torch.Tensor:
    kl = -0.5 * (1 + logvar - mu.pow(2) - logvar.exp()).sum(dim=-1)
    return kl.mean()

import torch


def make_alphas_cumprod(T: int, beta_start: float = 1e-4, beta_end: float = 0.02) -> torch.Tensor:
    betas = torch.linspace(beta_start, beta_end, T)
    return torch.cumprod(1.0 - betas, dim=0)


def q_sample(x0: torch.Tensor, t: torch.Tensor, alphas_cumprod: torch.Tensor, noise: torch.Tensor) -> torch.Tensor:
    abar = alphas_cumprod[t].view(-1, *([1] * (x0.dim() - 1)))
    return abar.sqrt() * x0 + (1 - abar).sqrt() * noise

import torch
import torch.nn.functional as F


def ddpm_loss(model, x0: torch.Tensor, t: torch.Tensor, alphas_cumprod: torch.Tensor, noise: torch.Tensor) -> torch.Tensor:
    abar = alphas_cumprod[t].view(-1, *([1] * (x0.dim() - 1)))

=== ddim-step | DDIM Sampling Step | Medium
--- description
Implement one deterministic DDIM update (eta = 0).

`ddim_step(x_t, eps_pred, abar_t, abar_prev)` where `abar_t` and `abar_prev` are the cumulative alphas (floats or 0-dim tensors) at the current and the previous (less noisy) timestep:

Return `(x_prev, x0_pred)`.

import torch


def ddim_step(x_t: torch.Tensor, eps_pred: torch.Tensor, abar_t, abar_prev):
    abar_t = torch.as_tensor(abar_t, dtype=x_t.dtype)
    abar_prev = torch.as_tensor(abar_prev, dtype=x_t.dtype)
    x0_pred = (x_t - torch.sqrt(1 - abar_t) * eps_pred) / torch.sqrt(abar_t)
    x_prev = torch.sqrt(abar_prev) * x0_pred + torch.sqrt(1 - abar_prev) * eps_pred
    return x_prev, x0_pred

=== classifier-free-guidance | Classifier-Free Guidance | Easy
--- description
Implement `guided_noise(model, x_t, t, cond, null_cond, scale)`.

The model is called as `model(x, t, c)` and returns a noise prediction shaped like `x`. Classifier-free guidance combines a conditional and an unconditional prediction:

```
eps = eps_uncond + scale * (eps_cond - eps_uncond)
```

where `eps_cond = model(x_t, t, cond)` and `eps_uncond = model(x_t, t, null_cond)`.

To save a forward pass, real implementations batch the two calls. Do the same: call the model exactly once on inputs of batch size `2B` (concatenate `x_t` with itself, `t` with itself and `cond` with `null_cond` along dim 0), then split the output.

import torch


def guided_noise(model, x_t: torch.Tensor, t: torch.Tensor, cond: torch.Tensor, null_cond: torch.Tensor, scale: float) -> torch.Tensor:
    x = torch.cat([x_t, x_t], dim=0)
    tt = torch.cat([t, t], dim=0)
    c = torch.cat([cond, null_cond], dim=0)
    eps_cond, eps_uncond = model(x, tt, c).chunk(2, dim=0)
    return eps_uncond + scale * (eps_cond - eps_uncond)

import torch
import torch.nn.functional as F


def discriminator_loss(real_logits: torch.Tensor, fake_logits: torch.Tensor) -> torch.Tensor:
    real = F.binary_cross_entropy_with_logits(real_logits, torch.ones_like(real_logits))
    fake = F.binary_cross_entropy_with_logits(fake_logits, torch.zeros_like(fake_logits))
    return real + fake


def generator_loss(fake_logits: torch.Tensor) -> torch.Tensor:
    return F.binary_cross_entropy_with_logits(fake_logits, torch.ones_like(fake_logits))

import torch
import torch.nn.functional as F


def flow_matching_loss(model, x0: torch.Tensor, x1: torch.Tensor, t: torch.Tensor) -> torch.Tensor:
    tb = t.view(-1, *([1] * (x0.dim() - 1)))
    x_t = (1 - tb) * x0 + tb * x1
    return F.mse_loss(model(x_t, t), x1 - x0)

=== discounted-returns | Discounted Returns | Easy
--- description
Implement `discounted_returns(rewards, gamma, dones=None)`.

- `rewards`: `(T,)` rewards of one rollout
- `dones`: optional `(T,)` tensor of 0/1 (or bool) flags; `dones[t] = 1` means the episode ended at step `t`, so no reward from later steps flows back past it


import torch


def discounted_returns(rewards: torch.Tensor, gamma: float, dones=None) -> torch.Tensor:
    rewards = rewards.float()
    T = rewards.shape[0]
    out = torch.zeros(T)
    running = 0.0
    for t in reversed(range(T)):
        not_done = 1.0 if dones is None else 1.0 - float(dones[t])
        running = rewards[t].item() + gamma * not_done * running
        out[t] = running
    return out

--- solution
import torch


def gae(rewards, values, next_value, dones, gamma: float = 0.99, lam: float = 0.95):
    T = rewards.shape[0]
    values = values.float()
    adv = torch.zeros(T)
    last = 0.0
    for t in reversed(range(T)):
        not_done = 1.0 - float(dones[t])
        v_next = float(next_value) if t == T - 1 else values[t + 1].item()
        delta = rewards[t].item() + gamma * v_next * not_done - values[t].item()
        last = delta + gamma * lam * not_done * last
        adv[t] = last
    return adv, adv + values

import torch


def ppo_clip_loss(logp_new: torch.Tensor, logp_old: torch.Tensor, advantages: torch.Tensor, clip_eps: float = 0.2) -> torch.Tensor:
    advantages = advantages.detach()
    ratio = torch.exp(logp_new - logp_old.detach())
    surr1 = ratio * advantages
    surr2 = torch.clamp(ratio, 1 - clip_eps, 1 + clip_eps) * advantages
    return -torch.min(surr1, surr2).mean()

import torch


def q_learning_update(Q: torch.Tensor, s: int, a: int, r: float, s_next: int, done: bool, alpha: float = 0.1, gamma: float = 0.99) -> torch.Tensor:
    target = r if done else r + gamma * Q[s_next].max().item()
    Q[s, a] += alpha * (target - Q[s, a].item())
    return Q


def epsilon_greedy(Q: torch.Tensor, s: int, epsilon: float, u: float, random_action: int) -> int:
    if u < epsilon:
        return random_action
    return int(Q[s].argmax().item())

import torch
import torch.nn.functional as F


def reinforce_loss(logits: torch.Tensor, actions: torch.Tensor, returns: torch.Tensor, baseline=None) -> torch.Tensor:
    logp = F.log_softmax(logits, dim=-1).gather(1, actions.unsqueeze(1)).squeeze(1)
    weights = returns if baseline is None else returns - baseline
    return -(logp * weights.detach()).mean()
